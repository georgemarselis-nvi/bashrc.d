AJAX ENDPOINTS (session auth required - no Bearer token)
#####################################################################################
# Auth: POST /login with application/x-www-form-urlencoded (username + password).
# Returns JSESSIONID cookie on 302 redirect. No CSRF preflight required.
# Use cookie jar for subsequent requests.
#
# POST body format for table/search endpoints (AntTableRequest):
#   {"pagination":{"current":1,"pageSize":10},"order":[{"property":"field","direction":"ascend"}],"search":[{"property":"field","value":"term"}]}
# SearchRequest adds: "global": true/false
#
# Source: confirmed from IRIDA source code (development branch). Not all endpoints tested empirically.

# AnalysesTableAjaxController - /ajax/analyses
# Confirmed working: /ajax/analyses/queue
GET  /ajax/analyses/queue                                           # Running/queued counts. Returns {"running":N,"queued":N}. Pure DB query, no Galaxy roundtrip. Safe to poll.
GET  /ajax/analyses/states                                          # AnalysisState enum values with localized labels. From source, not tested.
GET  /ajax/analyses/types                                           # Registered workflow types (assembly, phylogenomics, sistr, etc). From source, not tested.
POST /ajax/analyses/list                                            # Paginated analysis table. POST with AntTableRequest body. From source, not tested.
POST /ajax/analyses/delete                                          # Delete an analysis. From source, not tested.
GET  /ajax/analyses/download/{id}                                   # Download analysis outputs. From source, not tested.
GET  /ajax/analyses/{submissionId}/updated-table-progress           # Updated state and duration for a single submission. From source, not tested.

# AnalysesOutputsAjaxController - /ajax/analyses-outputs
GET  /ajax/analyses-outputs/shared?projectId={id}                   # Shared single sample outputs for a project. From source, not tested.
GET  /ajax/analyses-outputs/automated?projectId={id}                # Automated single sample outputs for a project. From source, not tested.
GET  /ajax/analyses-outputs/user                                    # All analysis outputs for current user (no project filter). From source, not tested.
POST /ajax/analyses-outputs/download/prepare                        # Prepare batch download selection. POST with list of output info. From source, not tested.
GET  /ajax/analyses-outputs/download/selection                      # Download prepared batch. From source, not tested.
GET  /ajax/analyses-outputs/download/file?analysisSubmissionId=&fileId=  # Download individual output file. From source, not tested.

# AnalysisAjaxController - /ajax/analysis
PATCH /ajax/analysis/update-email-pipeline-result                   # Toggle email notifications for pipeline completion/error. From source, not tested.
GET   /ajax/analysis/details/{submissionId}                         # Analysis details (workflow, version, priority, duration, share status). From source, not tested.
GET   /ajax/analysis/inputs/{submissionId}                          # Input files and samples for an analysis. From source, not tested.
PATCH /ajax/analysis/update-analysis                                # Update analysis name or priority. From source, not tested.
GET   /ajax/analysis/{id}/outputs                                   # List output files with metadata (size, tool, extension, first line). From source, not tested.
GET   /ajax/analysis/{id}/outputs/{fileId}                          # Read output file contents (lines or byte chunks via seek/chunk params). From source, not tested.
GET   /ajax/analysis/{submissionId}/job-errors                      # Galaxy job errors with Galaxy URL and history ID. From source, not tested.
GET   /ajax/analysis/{submissionId}/share                           # List projects an analysis can be shared with. From source, not tested.
POST  /ajax/analysis/{submissionId}/share                           # Share/unshare analysis with a project. From source, not tested.
POST  /ajax/analysis/{submissionId}/save-results                    # Save analysis results back to samples. From source, not tested.
GET   /ajax/analysis/sistr/{id}                                     # SISTR typing results. From source, not tested.
GET   /ajax/analysis/delete/{analysisSubmissionId}                   # Delete an analysis submission. From source, not tested.
GET   /ajax/analysis/download/{analysisSubmissionId}                 # Download all outputs as zip. From source, not tested.
GET   /ajax/analysis/{submissionId}/newick                          # Newick tree file contents. From source, not tested.
GET   /ajax/analysis/{submissionId}/image                           # Base64 encoded image output. From source, not tested.
GET   /ajax/analysis/{submissionId}/metadata                        # Sample metadata for analysis. From source, not tested.
GET   /ajax/analysis/{submissionId}/metadata-templates              # Metadata templates for analysis. From source, not tested.
GET   /ajax/analysis/{submissionId}/metadata-template-fields?templateId=  # Fields for a metadata template. From source, not tested.
GET   /ajax/analysis/{submissionId}/tree                            # Newick tree with validation. From source, not tested.
GET   /ajax/analysis/{submissionId}/parseExcel?filename=&sheetIndex=  # Parse excel output file. From source, not tested.
GET   /ajax/analysis/{submissionId}/provenance?filename=            # Tool execution provenance chain. From source, not tested.
GET   /ajax/analysis/{submissionId}/updated-progress                # Current state and duration. From source, not tested.
GET   /ajax/analysis/{submissionId}/analysis-details                # Full analysis info with sample/project associations. From source, not tested.

# SearchController - /ajax/search
POST /ajax/search/projects                                          # Search projects. POST with SearchRequest body. From source, not tested.
POST /ajax/search/samples                                           # Search samples. POST with SearchRequest body. From source, not tested.

# ProjectsController - /projects/ajax
GET  /projects/ajax/taxonomy/search?searchTerm=                     # Taxonomy search. From source, not tested.
GET  /projects/ajax/export?dtf=xlsx|csv&admin=                      # Export projects as xlsx or csv. From source, not tested.

# ProjectsAjaxController - /ajax/projects
